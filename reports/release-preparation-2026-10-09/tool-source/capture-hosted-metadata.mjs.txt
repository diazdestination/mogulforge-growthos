const fly='C:/Users/diazd/.fly/bin/fly.exe';
const safeString=(v,max=256)=>typeof v==='string'&&v.length<=max&&!/[\r\n\x00]/.test(v)?v:null;
const booleanState=v=>v===undefined?'unset':v===true||v==='true'?'true':v===false||v==='false'?'false':'unrecognized';
function rows(value,keys){if(Array.isArray(value))return value;for(const key of keys)if(Array.isArray(value?.[key]))return value[key];return null;}
function shape(value){return {recognized:false,kind:Array.isArray(value)?'array':typeof value,keys:value&&typeof value==='object'?Object.keys(value).slice(0,30):[]};}
function run(args){const startedAt=new Date().toISOString();return new Promise(resolve=>execFile(fly,args,{windowsHide:true,encoding:'utf8',timeout:25000,maxBuffer:2**21},(error,stdout)=>{if(error)return resolve({ok:false,startedAt,finishedAt:new Date().toISOString(),exitCode:Number.isInteger(error.code)?error.code:null,failure:'CLI inventory unavailable; raw error withheld'});try{resolve({ok:true,startedAt,finishedAt:new Date().toISOString(),value:JSON.parse(stdout)});}catch{resolve({ok:false,startedAt,finishedAt:new Date().toISOString(),failure:'CLI response was not JSON'});}}));}
function project(result,fn){if(!result.ok)return result;return {ok:true,startedAt:result.startedAt,finishedAt:result.finishedAt,...fn(result.value)};}
function machines(value){const list=rows(value,['machines','Machines']);if(!list)return shape(value);return {recognized:true,machines:list.map(m=>{const env=m.config?.env??{},service=m.config?.services?.[0]??{};const holds=Object.fromEntries(['GROWTHOS_OUTBOUND_HOLD','STAGING_OUTBOUND_APPROVED','SCHEDULER_AUTOSTART','AUTO_MIGRATE','STAGING_SCHEMA_PUSH_APPROVED','WEB_PUSH_ENABLED','GROWTHOS_RESTORE_ONLY_MODE'].map(key=>[key,booleanState(env[key])]));return {id:safeString(m.id),state:safeString(m.state),region:safeString(m.region),image:safeString(m.config?.image,512),imageDigest:safeString(m.image_ref?.digest,100),createdAt:safeString(m.created_at),updatedAt:safeString(m.updated_at),guest:m.config?.guest?{cpuKind:safeString(m.config.guest.cpu_kind),cpus:m.config.guest.cpus??null,memoryMb:m.config.guest.memory_mb??null}:null,holds,service:{autostart:service.autostart??null,autostop:service.autostop??null,minMachinesRunning:service.min_machines_running??null},emailAllowlistEmpty:env.STAGING_EMAIL_RECIPIENT_ALLOWLIST==='',appOrigin:env.APP_ORIGIN==='https://mogulforge-growthos-staging.fly.dev'||env.APP_ORIGIN==='https://app.mogulforge.ai'?env.APP_ORIGIN:null,cordoned:m.config?.metadata?.['fly_platform_version']?null:null};})};}
function secrets(value){const list=rows(value,['secrets','Secrets']);if(!list)return shape(value);return {recognized:true,names:list.map(s=>safeString(s.Name??s.name)).filter(Boolean).sort(),valuesRead:false,warning:'Secret names do not prove their values; effective environment override must be reconciled privately.'};}
function releases(value){const list=rows(value,['releases','Releases']);if(!list)return shape(value);return {recognized:true,releases:list.slice(0,5).map(r=>({version:r.Version??r.version??null,status:safeString(r.Status??r.status),createdAt:safeString(r.CreatedAt??r.created_at),image:safeString(r.ImageRef??r.image_ref??r.image,512)}))};}
function clusters(value){const list=rows(value,['clusters','Clusters']);if(!list)return shape(value);return {recognized:true,clusters:list.map(c=>({id:safeString(c.id??c.ID),name:safeString(c.name??c.Name),status:safeString(c.status??c.Status),region:safeString(c.region??c.Region),plan:safeString(c.plan??c.Plan),postgresMajorVersion:c.postgres_major_version??null}))};}
function clusterStatus(value){const c=value?.cluster??value;if(!c||typeof c!=='object'||Array.isArray(c))return shape(value);const attachments=c.attached_apps;return {recognized:!!(c.id??c.ID),id:safeString(c.id??c.ID),name:safeString(c.name??c.Name),status:safeString(c.status??c.Status),region:safeString(c.region??c.Region),plan:safeString(c.plan??c.Plan),postgresMajorVersion:c.postgres_major_version??null,attachedApps:Array.isArray(attachments)?attachments.map(a=>safeString(typeof a==='string'?a:a.name??a.app_name)).filter(Boolean):null,attachmentMeaning:Array.isArray(attachments)?'explicit array':'unknown; absence is not empty',availableKeys:Object.keys(c).slice(0,30)};}
function backups(value){const list=rows(value,['backups','Backups']);if(!list)return shape(value);return {recognized:true,backups:list.slice(0,12).map(b=>({id:safeString(b.id??b.backup_id??b.ID),label:safeString(b.name??b.label??b.Name),status:safeString(b.status??b.Status),type:safeString(b.type??b.backup_type),createdAt:safeString(b.created_at??b.CreatedAt),completedAt:safeString(b.completed_at??b.finished_at),size:b.size??b.size_bytes??null})),interpretation:'Metadata inventory only; no new backup requested and no managed restore or recovery proof.'};}
const apps=['mogulforge-growthos-staging','mogulforge-growthos'];
const appRecords=[];for(const app of apps){const [m,s,r]=await Promise.all([run(['machine','list','--app',app,'--json']),run(['secrets','list','--app',app,'--json']),run(['releases','--app',app,'--json'])]);appRecords.push({app,machines:project(m,machines),secrets:project(s,secrets),releases:project(r,releases)});}
const [c,b,t]=await Promise.all([run(['mpg','list','--json']),run(['mpg','backup','list','zp2wjrejd2zodn4q','--json']),run(['mpg','status','zp2wjrejd2zodn4q','--json'])]);
const result={kind:'READ_ONLY_HOSTED_CLI_METADATA_NO_SQL',recordedAt:new Date().toISOString(),apps:appRecords,clusterInventory:project(c,clusters),stagingCluster:project(t,clusterStatus),stagingBackupInventory:project(b,backups),hostedSql:false,mutations:false,rawProviderJsonStored:false,databaseCatalog:'NOT_REFRESHED; historical95 only until a new reviewed read-only observation',note:'Only public allowlisted configuration fields and secret names; not current effective secrets, schema, recovery or deployment approval.'};
fs.mkdirSync(path.dirname(destination),{recursive:true});fs.writeFileSync(destination,JSON.stringify(result,null,2)+'\n',{flag:'wx'});console.log(JSON.stringify(result));
